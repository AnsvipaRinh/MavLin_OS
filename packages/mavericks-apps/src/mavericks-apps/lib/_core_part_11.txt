def is_valid_key(key):
    if not isinstance(key, str) or not key:
        return False
    if _KEY_CHAR_RE.match(key) or _KEY_FN_RE.match(key):
        return True
    if key in NAMED_KEYS or key in MEDIA_KEYS:
        return True
    return False
